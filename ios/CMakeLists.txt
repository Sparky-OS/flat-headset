cmake_minimum_required(VERSION 3.16)
project(FlatHeadsetPattern LANGUAGES CXX)
enable_testing()
add_library(pattern Sources/Pattern.cpp)
target_include_directories(pattern PUBLIC Sources)
target_compile_features(pattern PUBLIC cxx_std_17)
target_compile_options(pattern PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(pattern-test Tests/PatternTest.cpp)
target_link_libraries(pattern-test PRIVATE pattern)
target_compile_options(pattern-test PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME pattern COMMAND pattern-test)

find_package(Threads REQUIRED)
add_executable(frames-test Session/Frames.cpp Tests/FramesTest.cpp)
target_compile_features(frames-test PRIVATE cxx_std_17)
target_compile_options(frames-test PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
target_link_libraries(frames-test PRIVATE Threads::Threads)
add_test(NAME decoded-frames COMMAND frames-test)

add_subdirectory(Viewer)
add_executable(viewer-test Tests/ViewerTest.cpp)
target_link_libraries(viewer-test PRIVATE viewer)
target_compile_options(viewer-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME viewer COMMAND viewer-test)

option(FH_VIEWER_GLES_TEST "Compare Android's GLES shader on a headless Linux renderer" OFF)
if(FH_VIEWER_GLES_TEST)
  add_executable(viewer-gles-test Tests/ViewerGlesTest.cpp)
  target_compile_features(viewer-gles-test PRIVATE cxx_std_17)
  target_compile_options(viewer-gles-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(viewer-gles-test PRIVATE EGL GLESv2)
  add_test(NAME viewer-android-gles COMMAND viewer-gles-test ${CMAKE_CURRENT_SOURCE_DIR}/../android/src/main/assets/headset_fitted.frag)
  set_tests_properties(viewer-android-gles PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1;LP_NUM_THREADS=4;GALLIUM_DRIVER=llvmpipe")
endif()

option(FH_VIEWER_SESSION_TEST "Check paired viewer descriptors with host OpenSSL" OFF)
if(FH_VIEWER_SESSION_TEST)
  find_package(OpenSSL REQUIRED)
  add_executable(viewer-session-test Tests/ViewerSessionTest.cpp Session/SessionControl.cpp Pairing/Pairing.cpp)
  target_compile_features(viewer-session-test PRIVATE cxx_std_17)
  target_compile_options(viewer-session-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(viewer-session-test PRIVATE OpenSSL::SSL OpenSSL::Crypto)
  add_test(NAME viewer-paired-panel COMMAND viewer-session-test)
endif()
