cmake_minimum_required(VERSION 3.28)
project(FlatHeadsetCore LANGUAGES C CXX)

if(NOT IS_DIRECTORY "${FREERDP_SOURCE_DIR}")
  message(FATAL_ERROR "FREERDP_SOURCE_DIR must name the verified FreeRDP source tree")
endif()
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "Inspect native archive members" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static receiver libraries" FORCE)
set(WITH_CLIENT_COMMON ON CACHE BOOL "Client embedding API" FORCE)
set(WITH_CHANNELS ON CACHE BOOL "RDP graphics transport" FORCE)
set(WITH_CLIENT_CHANNELS ON CACHE BOOL "Client channels" FORCE)
foreach(feature CLIENT CLIENT_SDL SERVER SAMPLE MANPAGES SYSTEMD X11 WAYLAND
                WINPR_TOOLS FUSE ALSA PULSE OSS MACAUDIO OPENH264 DSP_FFMPEG
                VAAPI VAAPI_H264_ENCODING GSM LAME FAAD2 FAAC SOXR AOM YUV
                OPUS JPEG CUPS PCSC SMARTCARD_EMULATE KRB5 URIPARSER AAD
                SSO_MIB CCACHE CLANG_FORMAT)
  set(WITH_${feature} OFF CACHE BOOL "Not used by the receiver core" FORCE)
endforeach()
set(WINPR_UTILS_IMAGE_PNG OFF CACHE BOOL "No image utilities" FORCE)
set(WINPR_UTILS_IMAGE_JPEG OFF CACHE BOOL "No image utilities" FORCE)
set(WINPR_UTILS_IMAGE_WEBP OFF CACHE BOOL "No image utilities" FORCE)
set(WITH_JSON_DISABLED ON CACHE BOOL "No Azure logon dependency" FORCE)
set(WITH_OPENSSL ON CACHE BOOL "TLS and authentication" FORCE)
set(OPENSSL_USE_STATIC_LIBS ON CACHE BOOL "Static target OpenSSL" FORCE)
set(WITH_INTERNAL_RC4 ON CACHE BOOL "Protocol compatibility" FORCE)
set(WITH_INTERNAL_MD4 ON CACHE BOOL "Protocol compatibility" FORCE)
set(WITH_INTERNAL_MD5 ON CACHE BOOL "Protocol compatibility" FORCE)
set(WITH_FFMPEG ON CACHE BOOL "Video codecs" FORCE)
set(WITH_VIDEO_FFMPEG ON CACHE BOOL "Video codecs" FORCE)
set(WITH_SWSCALE ON CACHE BOOL "Codec colour conversion" FORCE)
set(WITH_VIDEOTOOLBOX ${APPLE} CACHE BOOL "Apple hardware decode integration" FORCE)
# Keep the channels needed for the receiver; device/audio forwarding is not part of this build.
file(GLOB channel_options "${FREERDP_SOURCE_DIR}/channels/*/ChannelOptions.cmake")
foreach(options IN LISTS channel_options)
  get_filename_component(channel_dir "${options}" DIRECTORY)
  get_filename_component(channel "${channel_dir}" NAME)
  string(TOUPPER "${channel}" channel)
  set(enabled OFF)
  if(channel MATCHES "^(DRDYNVC|RDPGFX|DISP|CLIPRDR)$")
    set(enabled ON)
  endif()
  set(CHANNEL_${channel} ${enabled} CACHE BOOL "Receiver channel set" FORCE)
  set(CHANNEL_${channel}_CLIENT ${enabled} CACHE BOOL "Receiver client channel" FORCE)
  set(CHANNEL_${channel}_SERVER OFF CACHE BOOL "No server channels" FORCE)
endforeach()
add_subdirectory("${FREERDP_SOURCE_DIR}" freerdp)
add_executable(core-link-check CoreLinkCheck.c)
target_include_directories(core-link-check PRIVATE
  "${FREERDP_SOURCE_DIR}/include" "${FREERDP_SOURCE_DIR}/winpr/include"
  "${CMAKE_CURRENT_BINARY_DIR}/freerdp/include" "${CMAKE_CURRENT_BINARY_DIR}/freerdp/winpr/include"
  "${OPENSSL_INCLUDE_DIR}" "${AVCODEC_INCLUDE_DIRS}")
target_link_libraries(core-link-check PRIVATE freerdp-client freerdp winpr)
if(APPLE)
  target_link_options(core-link-check PRIVATE -Wl,-no_adhoc_codesign)
  set_target_properties(core-link-check PROPERTIES XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
else()
  enable_testing()
  add_test(NAME core-lifecycle COMMAND core-link-check)
endif()
